CREATE OR ALTER PROCEDURE dbo.notes_checkout
    @POSTEDINDTOPROCESS VARCHAR(250),
    @POSTEDINDPROCESSING VARCHAR(1),
    @CLIENTCODE VARCHAR(250),
    @NOTEDATE DATETIME,
    @ELAPSEDTIME INT,
    @ROWS INT,
    @INCLUDEBLANK BIT,
    @WHITESPACECHARACTERS NVARCHAR(32)
AS
BEGIN
    SET NOCOUNT ON;

    -- Evitar que el caller prolongue los bloqueos después del procedimiento.
    IF @@TRANCOUNT <> 0 OR (@@OPTIONS & 2) = 2
        THROW 50001,
            'notes_checkout requires autocommit without an ambient transaction.',
            1;

    IF @POSTEDINDTOPROCESS IS NULL
       OR LEN(@POSTEDINDTOPROCESS) = 0
       OR @POSTEDINDPROCESSING IS NULL
       OR DATALENGTH(@POSTEDINDPROCESSING) = 0
       OR @CLIENTCODE IS NULL
       OR LEN(@CLIENTCODE) = 0
       OR @ELAPSEDTIME IS NULL
       OR @ELAPSEDTIME <= 0
       OR @ROWS IS NULL
       OR @ROWS <= 0
       OR @INCLUDEBLANK IS NULL
       OR @WHITESPACECHARACTERS IS NULL
       OR DATALENGTH(@WHITESPACECHARACTERS) = 0
        THROW 50002, 'Invalid note checkout parameters.', 1;

    -- La reserva no puede ser uno de los valores elegibles.
    IF EXISTS (
        SELECT 1
        FROM STRING_SPLIT(@POSTEDINDTOPROCESS, ',')
        WHERE LTRIM(RTRIM(value)) = @POSTEDINDPROCESSING
    )
    OR DATALENGTH(
        TRIM(
            @WHITESPACECHARACTERS
            FROM CONVERT(NVARCHAR(1), @POSTEDINDPROCESSING)
                COLLATE Latin1_General_100_BIN2
        )
    ) = 0
        THROW 50003,
            'The processing indicator must differ from eligible indicators and whitespace.',
            1;

    DECLARE @SelectedAt DATETIME = GETDATE();
    DECLARE @Cutoff DATETIME =
        DATEADD(MINUTE, -@ELAPSEDTIME, @SelectedAt);

    ;WITH EligibleNotes AS
    (
        SELECT N.*
        FROM dbo.tbl_Notes AS N WITH (UPDLOCK, HOLDLOCK)
        WHERE
        (
            N.PostedInd IN (
                SELECT LTRIM(RTRIM(value))
                FROM STRING_SPLIT(@POSTEDINDTOPROCESS, ',')
            )
            OR
            (
                @INCLUDEBLANK = 1
                AND DATALENGTH(N.PostedInd) > 0
                AND DATALENGTH(
                    TRIM(
                        @WHITESPACECHARACTERS
                        FROM CONVERT(NVARCHAR(MAX), N.PostedInd)
                            COLLATE Latin1_General_100_BIN2
                    )
                ) = 0
            )
        )
        AND
        (
            N.FileNoteDateTime IS NULL
            OR N.FileNoteDateTime < @Cutoff
        )
        AND EXISTS
        (
            SELECT 1
            FROM dbo.tbl_LossDraftSummary AS S
            WHERE S.ReferenceID = N.ReferenceID
              AND S.ClientCode IN (
                  SELECT LTRIM(RTRIM(value))
                  FROM STRING_SPLIT(@CLIENTCODE, ',')
              )
        )
    ),
    SelectedGroups AS
    (
        SELECT TOP (@ROWS)
            ReferenceID,
            NoteDateTime
        FROM EligibleNotes
        GROUP BY ReferenceID, NoteDateTime
        ORDER BY NoteDateTime, ReferenceID
    )
    UPDATE E
    SET PostedInd = @POSTEDINDPROCESSING,
        FileNoteDateTime = @SelectedAt
    OUTPUT inserted.*
    FROM EligibleNotes AS E
    INNER JOIN SelectedGroups AS G
        ON (
            G.ReferenceID = E.ReferenceID
            OR (G.ReferenceID IS NULL AND E.ReferenceID IS NULL)
        )
       AND (
            G.NoteDateTime = E.NoteDateTime
            OR (G.NoteDateTime IS NULL AND E.NoteDateTime IS NULL)
        );
END;
GO
