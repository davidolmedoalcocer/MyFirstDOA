  /**
   * Finds a version in the page's version list. Matched by its `data-version` attribute, not by text:

  /**
   * Reads the Configurations table as the user sees it, top to bottom.
   *
   * Cells are matched to their column by header title, not by position. The empty-state row is not a
   * data row, so an empty table gives an empty list.
   * @returns One entry per visible row, with its Key and Value text.
   * @throws {Error} If the table has no Key or Value column.
   */
  async getConfigurationRows(): Promise<ConfigurationRow[]> {
    const table = this.configurationsTable.locator
    await table.waitFor()
    const titles = (await table.locator('thead th .header-title').allTextContents()).map((title) => title.trim())
    const keyColumn = titles.indexOf('Key')
    const valueColumn = titles.indexOf('Value')
    if (keyColumn < 0 || valueColumn < 0) {
      throw new Error(`The Configurations table needs Key and Value columns, but has: ${titles.join(', ')}.`)
    }

    const rows = table.locator('tbody tr:not(.empty-state-row)')
    const result: ConfigurationRow[] = []
    for (let index = 0; index < (await rows.count()); index++) {
      const cells = await rows.nth(index).locator('td').allTextContents()
      result.push({ key: (cells[keyColumn] ?? '').trim(), value: (cells[valueColumn] ?? '').trim() })
    }
    return result
  }

    await expect(this.page.locator('[data-testid="product-heading"]')).toHaveText(displayName)
  }

  /**
   * Picks a key in the Assign Configuration panel's Key selector.
   *
   * Matched as a substring, not anchored: an option whose key is too long for the list gets a tooltip in its
   * shadow DOM, which Playwright counts as part of the option's text, so a full-text match stops finding it.
   * Test keys end in a random suffix, so the substring still names one option, and the locator is strict:
   * two matches fail instead of picking the wrong one.
   * @param key - The Setting's key.
   */

    const option = this.assignConfigurationKeySelect.locator
      .locator('infios-select-option')

    await this.assignConfigurationKeySelect.locator.click()
    await expect(option).toBeVisible()

    await expect(this.assignConfigurationKeySelect.locator).toHaveJSProperty('value', await option.getAttribute('value'))
  }

  /**
   * Moves the pointer onto a Configuration's row, as a user does to reach its actions. The table renders
   * row actions only for the hovered row, so the pointer moves off the table first and then onto the row.
   * @param key - The Setting key shown in the row.
   * @returns The row.
   * @throws {Error} If the table has no row for the key.

  async hoverConfigurationRow(key: string): Promise<Locator> {
    const row = await this.configurationsTable.getRowByText(key)
    if (!row) throw new Error(`No row for key "${key}".`)
    await this.versionHeading.hover()
    await row.hover()
    return row
  }

  /*

  /** The drawer's title. */
  async getDrawerLabel(): Promise<string> {
    return this.drawer.locator.evaluate((el) => (el as unknown as { label: string }).label)

  /** The validation messages a field shows. */
  async getFieldErrors(label: ProductField): Promise<string[]> {
    return this.field(label).locator.evaluate((el) => [...((el as unknown as { errors?: string[] }).errors ?? [])])
  }

  async isFieldRequired(label: ProductField): Promise<boolean> {
    return this.field(label).locator.evaluate((el) => !!(el as unknown as { required?: boolean }).required)
  }

  /** Whether a field's input is read-only. */

      await this.input(label).setText(text)
