# Use the same UID and GID that Compose uses to run the UI services.
ui_user=$(docker compose -f ./docker-compose-devui.yml config --format json |
    node -p 'JSON.parse(require("fs").readFileSync(0, "utf8")).services["dev-portal-admin-ui"].user')
test -n "$ui_user"

# Build first so the repair also works when the UI images are missing.
docker compose -f ./docker-compose-devui.yml build

# Repair root-owned Vite cache entries left by older UI containers.
for service in dev-portal-admin-ui dev-portal-tenant-ui; do
    docker compose -f ./docker-compose-devui.yml run --rm --no-deps \
        --user 0 --entrypoint sh "$service" -ec '
            cache=node_modules/.vite
            if [ -d "$cache" ] && [ ! -L "$cache" ]; then
                find "$cache" -uid 0 -exec chown -h "$1" {} +
            fi
        ' sh "$ui_user"
done

docker compose -f ./docker-compose-devui.yml up -d
