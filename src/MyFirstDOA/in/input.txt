from pathlib import Path
import re

base = Path("/workspaces/dev-portal-513634")
files = {
    ".devcontainer/detect-and-write-volume.sh": 1,
    "docker-compose-devui.yml": 2,
}

changes = {}

for filename, expected in files.items():
    path = base / filename
    original = path.read_text()
    updated, count = re.subn(
        r"case 01",
        "EACCES",
        original,
        flags=re.IGNORECASE
    )

    if count != expected:
        raise SystemExit(
            f"DETENIDO: {filename}: esperaba {expected} referencias, encontré {count}. No se modificó ningún archivo."
        )

    changes[path] = updated

for path, updated in changes.items():
    path.write_text(updated)
    print(f"CORREGIDO: {path.name}")

print("PASS: 3 referencias corregidas")
