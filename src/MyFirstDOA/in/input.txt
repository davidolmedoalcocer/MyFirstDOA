SELECT p."Id", p."Name", p."DisplayName"
FROM "Product" AS p
ORDER BY p."DisplayName", p."Name";


SELECT s."Id", s."Key"
FROM "Setting" AS s
ORDER BY s."Key";


SELECT
    s."Key",
    sv."RevisionNumber",
    sv."ManifestPlacement",
    sv."ValueStrategy",
    sv."ValueExpression",
    sv."GeneratedFormat",
    sv."GeneratedLength"
FROM "Setting" AS s
JOIN "SettingVersion" AS sv ON sv."SettingId" = s."Id"
WHERE s."Key" = 'jwt.signing.key'
ORDER BY sv."RevisionNumber" DESC
LIMIT 1;


SELECT
    p."DisplayName" AS producto,
    pv."Id" AS id_version,
    pv."MajorVersion" AS mayor,
    pv."MinorVersion" AS menor,
    pv."PatchVersion" AS parche,
    s."Key" AS llave
FROM "ProductVersionSetting" AS pvs
JOIN "ProductVersion" AS pv ON pv."Id" = pvs."ProductVersionId"
JOIN "Product" AS p ON p."Id" = pv."ProductId"
JOIN "Setting" AS s ON s."Id" = pvs."SettingId"
WHERE s."Key" = 'jwt.signing.key'
ORDER BY p."DisplayName", mayor, menor, parche;
