  discrimina entre las causas 1, 3 y 7 a la vez:
  
  SELECT
      current_database()                                    AS db,
      inet_server_port()                                    AS puerto_servidor,
      (SELECT count(*) FROM information_schema.columns
         WHERE table_name = 'ClusterGroup'
           AND column_name = 'EnvironmentType')             AS tiene_environmenttype,
      (SELECT count(*) FROM databasechangelog
         WHERE id = 'add-clustergroup-environmenttype')     AS changeset_aplicado,
      (SELECT count(*) FROM information_schema.tables
         WHERE table_name = 'ProductBuild')                 AS tabla_productbuild,
      (SELECT count(*) FROM information_schema.tables
         WHERE table_name = 'ProductVersion')               AS tabla_productversion,
