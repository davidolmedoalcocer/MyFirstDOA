#!/bin/bash
set -e

# =============================================================================
# SDLC Framework Setup
# Handles: GitHub auth, Azure auth, Kiro CLI login check, framework install,
#          team config pull, UIKit steering files
# Run via VS Code task: "Setup SDLC Framework". Safe to re-run: each step
# skips work that is already done.
# =============================================================================

# The framework moved from GitHub Packages (@envistainteractive) to the ADO feed
# `infios-ai-sdlc`; its install page is docs/INSTALL.md in the archer-sdlc repo.
SDLC_INSTALLER_URL="https://dev.azure.com/highjumpvso/Archer Platform/_apis/git/repositories/archer-sdlc/items?path=install/install.sh"
TEAM_CONFIG_REPO="infios-tools/neptune-sdlc-config"
TEAM_CONFIG_BRANCH="master"
KIRO_IDENTITY_PROVIDER="https://d-9a671a9ffc.awsapps.com/start"
KIRO_REGION="us-east-1"
KIRO_DIR=".kiro"
SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"

# Colors for output
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m' # No Color

info()  { echo -e "${GREEN}✓${NC} $1"; }
warn()  { echo -e "${YELLOW}⚠${NC} $1"; }
error() { echo -e "${RED}✗${NC} $1"; }
step()  { echo -e "\n${GREEN}━━━ $1 ━━━${NC}"; }

# =============================================================================
# Step 1: GitHub CLI Authentication
# Needed to clone the team configuration repo in Step 5.
# =============================================================================
step "Step 1/6: GitHub Access"

TEAM_CONFIG_URL="https://github.com/$TEAM_CONFIG_REPO.git"

# VS Code forwards the host's Git credentials into the dev container (the same
# sign-in used to clone this repo), so git usually reaches GitHub with no extra
# login. Only fall back to the gh login when that doesn't work.
can_read_team_config() {
    GIT_TERMINAL_PROMPT=0 git ls-remote "$TEAM_CONFIG_URL" HEAD &>/dev/null
}

if can_read_team_config; then
    info "GitHub access works through VS Code's Git credentials"
else
    warn "Git can't reach $TEAM_CONFIG_REPO with VS Code's credentials — starting gh login..."
    echo "  gh prints a one-time code; enter it at https://github.com/login/device."
    echo "  Ignore any 'Failed to copy one-time code to clipboard' warning: the container has no clipboard."
    echo ""
    gh auth login --hostname github.com --git-protocol https --web
    gh auth setup-git
    if ! can_read_team_config; then
        error "Your GitHub account cannot read $TEAM_CONFIG_REPO."
        echo "  Ask a team lead for access to the infios-tools GitHub org, then re-run this task."
        exit 1
    fi
    info "GitHub CLI authenticated"
fi

# Make a token available to the GitHub MCP server: now, so `infios-sdlc init` in
# Step 4 writes the server entry, and in every new terminal through ~/.bashrc, so
# Kiro started from a terminal can use it.
source "$SCRIPT_DIR/github-token-env.sh"
BASHRC_LINE="[ -f /workspaces/dev-portal/.workspaceScripts/github-token-env.sh ] && source /workspaces/dev-portal/.workspaceScripts/github-token-env.sh"
if ! grep -qF "$BASHRC_LINE" "$HOME/.bashrc" 2>/dev/null; then
    echo "$BASHRC_LINE" >> "$HOME/.bashrc"
fi
if [ -n "$GITHUB_TOKEN" ]; then
    info "GitHub token exported for the GitHub MCP server (new terminals get it from ~/.bashrc)"
else
    warn "No GitHub token found; the GitHub MCP server will be skipped. Run 'gh auth login', then re-run this task."
fi

# =============================================================================
# Step 2: Azure CLI Authentication
# The only credential the framework installer needs (ADO feed infios-ai-sdlc).
# =============================================================================
step "Step 2/6: Azure DevOps Authentication"

# Uses the browser sign-in through az-login.sh, which fixes the redirect page that
# hangs (see the comment there). Device code login is blocked by IT policy.
# The token request is for the Azure DevOps resource, which is what the framework
# installer needs.
ADO_RESOURCE_ID="499b84ac-1321-427f-aa17-267ca6975798"
has_ado_token() { az account get-access-token --resource "$ADO_RESOURCE_ID" &>/dev/null; }

if has_ado_token; then
    info "Azure CLI is signed in with access to Azure DevOps"
else
    warn "Azure CLI is not signed in — opening a browser sign-in..."
    echo "  Sign in with your Infios account in the browser tab that opens."
    echo "  Infios accounts have no Azure subscriptions, so press Enter at the tenant prompt."
    echo ""
    bash "$(dirname "$0")/az-login.sh" || true
    if ! has_ado_token; then
        error "Azure sign-in did not complete."
        echo ""
        echo "  In a terminal in this container, run:"
        echo ""
        echo "      bash .workspaceScripts/az-login.sh"
        echo ""
        echo "  Then re-run this task."
        exit 1
    fi
    info "Azure CLI signed in"
fi

# =============================================================================
# Step 3: Kiro CLI Login
# Not needed by the rest of this script, so a failed login only warns.
# =============================================================================
step "Step 3/6: Kiro CLI Login"

if ! command -v kiro-cli &>/dev/null; then
    warn "kiro-cli not found. It is installed by .devcontainer/postCreateCommand.sh; rebuild the container or run:"
    echo "    curl -fsSL https://cli.kiro.dev/install | bash"
elif kiro-cli whoami &>/dev/null; then
    info "Kiro CLI already logged in"
else
    # Every Infios seat is a Pro tier through the same Identity Center; the tier
    # comes from the account's assignment, so one command works for everyone.
    # The device flow avoids a localhost redirect into the container. It is the
    # AWS device flow; the Microsoft sign-in it leads to is a normal browser login.
    KIRO_LOGIN="kiro-cli login --license pro --identity-provider $KIRO_IDENTITY_PROVIDER --region $KIRO_REGION --use-device-flow"
    warn "Kiro CLI is not logged in — starting login..."
    echo "  Open the URL it prints and confirm the code, then sign in with your Infios account."
    echo ""
    $KIRO_LOGIN || true
    if kiro-cli whoami &>/dev/null; then
        info "Kiro CLI logged in"
    else
        # Not needed by the rest of this script, so continue.
        warn "Kiro CLI login did not complete. Run it later in a terminal:"
        echo "    $KIRO_LOGIN"
    fi
fi

# =============================================================================
# Step 4: Install/Update SDLC Framework
# The installer installs the CLI from the `latest` channel, removes the retired
# @envistainteractive build, then runs `infios-sdlc init --kiro` in this repo.
# On an existing install, init runs `update` instead, so re-runs are safe.
# =============================================================================
step "Step 4/6: SDLC Framework Install"

# Download first so a failed fetch stops here instead of piping an error into sh.
INSTALLER=$(az rest --only-show-errors --resource https://app.vssps.visualstudio.com -u "$SDLC_INSTALLER_URL")
if [ -z "$INSTALLER" ]; then
    error "Could not download the framework installer from archer-sdlc."
    echo "  Check that your account can read the Archer Platform project in ADO."
    exit 1
fi

printf '%s\n' "$INSTALLER" | sh -s -- --kiro
hash -r
info "Framework installed: $(infios-sdlc --version)"

# =============================================================================
# Step 5: Pull Team Configuration
# =============================================================================
step "Step 5/6: Team Configuration"

TEAM_CONFIG_TMP="/tmp/neptune-sdlc-config"

if [ -d "$TEAM_CONFIG_TMP" ]; then
    git -C "$TEAM_CONFIG_TMP" fetch origin "$TEAM_CONFIG_BRANCH" --depth 1
    git -C "$TEAM_CONFIG_TMP" reset --hard "origin/$TEAM_CONFIG_BRANCH"
else
    git clone --depth 1 --branch "$TEAM_CONFIG_BRANCH" "$TEAM_CONFIG_URL" "$TEAM_CONFIG_TMP"
fi

# Remove .gitkeep files so they don't get copied into the working repo
find "$TEAM_CONFIG_TMP" -name ".gitkeep" -delete

# Overlay team customizations (don't clobber repo-specific files if they exist)
if [ -d "$TEAM_CONFIG_TMP/agent-overrides" ]; then
    cp -r "$TEAM_CONFIG_TMP/agent-overrides/" "$KIRO_DIR/"
    info "Copied agent-overrides"
fi

if [ -d "$TEAM_CONFIG_TMP/skills" ]; then
    cp -r "$TEAM_CONFIG_TMP/skills/" "$KIRO_DIR/"
    info "Copied skills"
fi

if [ -f "$TEAM_CONFIG_TMP/team-config.yaml" ]; then
    cp "$TEAM_CONFIG_TMP/team-config.yaml" "$KIRO_DIR/team-config.yaml"
    info "Copied team-config.yaml"
fi

# Merge repo-local overrides if present
if [ -f "$KIRO_DIR/team-config.local.yaml" ]; then
    yq eval-all 'select(fileIndex == 0) * select(fileIndex == 1)' \
        "$KIRO_DIR/team-config.yaml" "$KIRO_DIR/team-config.local.yaml" \
        > "$KIRO_DIR/team-config.merged.yaml"
    mv "$KIRO_DIR/team-config.merged.yaml" "$KIRO_DIR/team-config.yaml"
    info "Merged repo-local overrides from team-config.local.yaml"
fi

# Copy any additional team files (custom flows, gates, commands, etc.)
for dir in flows quality-gates commands swarm-patterns; do
    if [ -d "$TEAM_CONFIG_TMP/$dir" ]; then
        # Only copy files that don't already exist (additive, not overwriting framework)
        find "$TEAM_CONFIG_TMP/$dir" -type f | while read -r file; do
            rel="${file#$TEAM_CONFIG_TMP/}"
            if [ ! -f "$KIRO_DIR/$rel" ]; then
                mkdir -p "$KIRO_DIR/$(dirname "$rel")"
                cp "$file" "$KIRO_DIR/$rel"
            fi
        done
        info "Synced custom $dir"
    fi
done

# =============================================================================
# Step 6: UIKit Steering Files
# =============================================================================
step "Step 6/6: UIKit Steering Files"

# VS Code tasks run non-interactive bash, which skips ~/.bashrc. Source the
# artifacts-helper wrappers directly so pnpm gets ARTIFACTS_ACCESSTOKEN.
if [ -f "$HOME/artifacts-helper.sh" ]; then
    source "$HOME/artifacts-helper.sh"
fi

mkdir -p "$KIRO_DIR/steering"

# Install sharedUIComponents if no node_modules exist
if [ ! -d "sharedUIComponents/node_modules/@infios/uikit" ]; then
    info "Installing sharedUIComponents dependencies..."
    if ! (cd sharedUIComponents && pnpm install --frozen-lockfile); then
        error "pnpm install failed. A 401 here means the ADO package feeds are not authenticated:"
        echo "  sign in when the 'ADO Codespaces Auth' extension prompts, then re-run this task."
        exit 1
    fi
fi

UIKIT_STEERING="sharedUIComponents/node_modules/@infios/uikit/dist/steering"

if [ -d "$UIKIT_STEERING" ]; then
    cp "$UIKIT_STEERING"/*.md "$KIRO_DIR/steering/"
    info "Copied UIKit steering files from $UIKIT_STEERING"
else
    warn "UIKit steering files not found — @infios/uikit may have failed to install"
fi

# =============================================================================
# Verification
# =============================================================================
step "Verification"

echo ""
infios-sdlc check || true
echo ""
info "SDLC framework setup complete!"
echo ""
echo "  Next steps:"
echo "    • Open Kiro and run /neptune:setup-check for full MCP verification"
echo "    • Run /neptune:help to see available commands"
echo ""
